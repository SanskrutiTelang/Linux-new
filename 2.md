#!/bin/bash

# Generate ~1MB of text-heavy data into data.txt
# Each line is ~120 characters of words, so ~8,500 lines ≈ 1MB

output="data.txt"
lines=8500

> "$output"  # clear file if it exists:wq


for i in $(seq 1 $lines); do
  echo "The quick brown fox jumps over the lazy dog again and again, creating a stream of words for testing purposes lorem ipsum dolor sit amet." >> "$output"
done

# Show file size
ls -lh "$output"
